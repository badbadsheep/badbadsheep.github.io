这时就可以使用格式控制字符串了，如：
`printf("%4lld", ...)`（在%和lld之间加入入一个4，表示打印出的整数如果小于4位数，则在左边补空格直到列宽为4）


#### 函数

函数语法：

长相：<输入的类型> + <函数名（随便起）> + <形式参数>

声明与定义可以这么总结：

1.在main函数前，定义就相当于声明了

2.定义如果在main后面，则必须在main前面声明

3.满足1点2点后，在任何地方都可以使用函数（甚至是定义之前！因为编译的时候main就已经认识新函数了！）

编写一个函数过程，判定一个数是否为“回文数”。所谓“回文数”是指对称位置上的数字相同，如121，2332等。


```c
/*先要确定判断回文数的方案————
是靠提取对称位置的数字进行比对，还是利用数学性质？c语言确实可以拆字符串，但是没有运用数学性质那么方便和优雅
观察回文数的性质：123321 发现肉眼上的对称有点难转化成严谨的数学关系。于是换个方式思考：

直接设计一个把数字倒过来的函数 再和输入的数字匹配————
依次对百位 十位 个位取余？可怎么知道有几位呢？这是把思路限制在了“从左往右的阅读习惯上”，完全可以从右往左拆！
先对10取余，就得到了个位数；再除以10抛掉0，接着对10取余得到十位数...直到最后得到了0！(这是一个循环次数未知的结构，用while！)

比如321--> 1 ---> 2 ---> 3  同时，每得到一个数字就乘以10再加下一个数 从而拼起来
如下：
*/
int judge(int a){
    int result = 0;
    while(a!=0){
            
            int b = a % 10 ;   // b用来储存得到的一个数字
            a = (a-b)/10 ;     // 删掉a的个位数   
            result = result*10 + b;   //乘以10再加上新的
         
    }
    return result;
}

# include <stdio.h>
int main(){
    int a = 0,b = 0;
    scanf("%d",&a);
    b = judge(a);
    printf("%d\n",b);
    if(a==b){
        printf("是回文数");
    }else{
        printf("不是回文数");
    }
return 0;
}
```

Q4
计算调和级数，输入x，输出前x项之和（准确的分数形式），如：【样例输入】10 【样例输出】7381/2520
```c
/*由于最终结果要是精准的分数形式，就不能直接做除法，而是通分再加起来
所以问题的关键在于求分母，即最小公倍数！先自己想象一个思路：
    拿最大的数n，依次求它的1倍，2倍...每求一次，就循环拿n除以1 2 3...只要有除不尽的就换下一个n的倍数。全部除尽就取该值
*/
# include <stdio.h>

long long LCM(int n){
    for (long long k=1;k>=1;k++){  //k是几倍的n
        for (long long i=1;i<=n;i++){  //依次拿kn去挨个除以1 2 3 4 5 ...(不要直接写kn了，是k*n！)
            long long q = k*n % i;
            if (q!=0)     //只要余数不为0就中断
                break;
            if (q==0 && i==n){         //能坚持除到第n项就暂停并返回值
                long long result=k*n;
                return result;       
            }    
        if (i == n)
            return result;
        }
    }
}

int main(){
    int n=0;
    scanf("%lld",&n);
    long long b=LCM(n);
    printf("%lld",b);
}
```
